cmake_minimum_required(VERSION 3.20)
project(service_api LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(SQLite3 REQUIRED)
include(FetchContent)
FetchContent_Declare(httplib URL https://github.com/yhirose/cpp-httplib/archive/refs/tags/v0.18.3.tar.gz)
set(HTTPLIB_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(httplib)
add_executable(service-api main.cpp)
target_link_libraries(service-api PRIVATE httplib::httplib SQLite::SQLite3)
